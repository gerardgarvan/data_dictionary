%macro p7_check_manifest;
  %local _p7m_modate _p7m_nobs _p7m_cur_modate _p7m_cur_nobs _p7m_file;
  %let _p7m_file = &exchange_path.\profile_PCNR_manifest.csv;

  /* Fail if manifest does not exist */
  %if %sysfunc(fileexist(&_p7m_file)) = 0 %then %do;
    %put ERROR: P7-MANIFEST: profile_PCNR_manifest.csv not found -- rerun Phase 6 first.;
    %let _p7_fail = 1;
    %return;
  %end;

  /* Read manifest values */
  data _null_;
    infile "&_p7m_file" dsd firstobs=2 truncover;
    input pcnr_modate :best32. pcnr_nobs :best32. run_dttm $20.;
    call symputx('_p7m_modate', pcnr_modate, 'L');
    call symputx('_p7m_nobs',   pcnr_nobs,   'L');
  run;

  /* Get current pdd.pcnr modate and nobs */
  proc sql noprint;
    select modate, nobs
    into :_p7m_cur_modate trimmed, :_p7m_cur_nobs trimmed
    from dictionary.tables
    where libname='PDD' and memname='PCNR';
  quit;

  %if &_p7m_cur_modate = %then %do;
    %put ERROR: P7-MANIFEST: pdd.pcnr not found in dictionary.tables.;
    %let _p7_fail = 1;
    %return;
  %end;

  /* Compare modate and nobs */
  %if %sysevalf(&_p7m_modate ne &_p7m_cur_modate) %then %do;
    %put ERROR: P7-MANIFEST: manifest pcnr_modate (&_p7m_modate) differs from current pdd.pcnr modate (&_p7m_cur_modate). Rerun Phase 6.;
    %let _p7_fail = 1;
  %end;
  %if %sysevalf(&_p7m_nobs ne &_p7m_cur_nobs) %then %do;
    %put ERROR: P7-MANIFEST: manifest pcnr_nobs (&_p7m_nobs) differs from current pdd.pcnr nobs (&_p7m_cur_nobs). Rerun Phase 6.;
    %let _p7_fail = 1;
  %end;
%mend p7_check_manifest;

%macro p7_check_namemap_fresh;
  %local _p7n_modate _p7n_csvdate _p7n_fid _p7n_mdate_str;
  proc sql noprint;
    select modate
    into :_p7n_modate trimmed
    from dictionary.tables
    where libname='PDD' and memname='NAME_MAP';
  quit;
  %if &_p7n_modate = %then %do;
    %put ERROR: P7: pdd.name_map not found.;
    %let _p7_fail = 1;
    %return;
  %end;
  /* Get CSV last-modified via fileinfo */
  filename _p7nm "&exchange_path.\name_map.csv";
  %let _p7n_fid = %sysfunc(fopen(_p7nm));
  %if &_p7n_fid <= 0 %then %do;
    %put ERROR: P7: name_map.csv not found.;
    %let _p7_fail = 1;
    filename _p7nm clear;
    %return;
  %end;
  %let _p7n_mdate_str = %sysfunc(finfo(&_p7n_fid, Last Modified));
  %let _p7n_fid = %sysfunc(fclose(&_p7n_fid));
  filename _p7nm clear;
  /* Convert: modate is SAS datetime; CSV date is OS locale string */
  /* Compare only date part */
  %let _p7n_csvdate = %sysfunc(inputn(%scan(&_p7n_mdate_str,1,%str( )), anydtdte10.));
  %if %sysevalf(&_p7n_csvdate < %sysfunc(datepart(&_p7n_modate))) %then %do;
    %put ERROR: P7: name_map.csv is older than pdd.name_map -- rerun Phase 4 export.;
    %let _p7_fail = 1;
  %end;
%mend p7_check_namemap_fresh;
%p7_check_namemap_fresh;
